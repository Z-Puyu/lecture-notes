        \begin{itemize}
            \setlength\itemsep{0.1em}
            \item \textbf{Co-finite}: open $\iff$ \textbf{complements of finite sets}.
            \item \textbf{Co-countable}: open $\iff$ \textbf{complements of countable sets}.
            \item $\mathcal{B}$ gen. the coarsest containing itself.
            \item \textbf{Metric}:  $\geq 0$, \textbf{definite, symmetric, $\triangle$-inequality}.
            \item \textbf{Pseudo-metric}: $d\left(x, x\right) = 0$ but can be not definite.
            \item \textbf{Quasi-metric}: can be not symmetric.
            \item \textbf{Norm}: $\geq 0$, \textbf{definite, $\triangle$-inequality}, $\norm{\lambda x} = \abs{\lambda}\norm{x}$.
            \item \textbf{Distance between sets}: smallest pointwise distance.
            \item \textbf{Diameter of set}: $\sup$ of pointwise distance.
            \item $L^p$\textbf{-metric}: gen. the standard topo. on $\R^n$.
            \begin{equation*}
                \max\norm{y_i - x_i} \leq \left[\sum_{i = 1}^{n}\norm{y_i - x_i}^p\right]^{\frac{1}{p}} \leq n^{\frac{1}{p}}\max\norm{y_i - x_i}.
            \end{equation*}
            \item Metrics are equivalent $\iff c_1d \leq d' \leq c_2d$.
            \item $x \in \overline{A} \iff \forall$ open neighbourhood $U$ of $x$, $U \cap A \neq \varnothing$.
            \item \textbf{Continuity}: $f^{-1}\left(S\right)$ is open $\forall S$ in sub-basis.
            \item TFAE:
            \begin{enumerate}
                \item $f$ is cont.;
                \item for all $A \subseteq X$, $f\left(\overline{A}\right) \subseteq \overline{f\left(A\right)}$;
                \item for any closed set $B \subseteq Y$, $f^{-1}\left(B\right)$ is closed in $X$;
                \item $\forall x \in X$ and any open $V \subseteq Y$ with $f\left(x\right) \in V$, there is an open set $U \subseteq X$ s.t. $x \in U$ and $f\left(U\right) \subseteq V$.
            \end{enumerate}
            \item \textbf{Dense}: $\forall$ open $U$ containing $x \in X$, $U \cap A \neq \varnothing$.
            \item Dense $\iff \mathring{X \setminus A} = \varnothing \iff \overline{A} = X$.
            \item \textbf{Pasting lemma}: if $X = A \cup B$ for closed $A, B$ and $f = g$ for all $x \in A \cap B$, then $h = x \in A ? f : \left(x \in B ? g\right)$ is cont. if $f \colon A \to Y$ and $g \colon B \to Y$ are.
            \item \textbf{Pull-back} topo.: $\left\{f^{-1}\left(U\right) \colon U \textrm{ is open}\right\}$ is the coarsest topo. ensuring cont. $f$. (quotient)
            \item \textbf{Uniform continuity}: $\forall \epsilon > 0$, there exists some $\delta > 0$ s.t. $d_X\left(x, y\right) < \delta \implies d_Y\bigl(f\left(x\right), f\left(y\right)\bigr) < \epsilon$.
            \item $f$ is uniformly cont. $\iff \forall \left\{x_i\right\}_i^{\infty}, \left\{y_i\right\}_{i}^{\infty}$ s.t. $\lim_{i \to \infty}d_X\left(x_i, y_i\right) = 0$, $\lim_{i \to \infty}d_Y\bigl(f\left(x_i\right), f\left(y_i\right)\bigr) = 0$.
            \item Limit of uniformly convergent sequence is cont..
            \item $x \mapsto x^n$ converges to $\mathbf{1}_{x = 1}$ pointwise but not uniformly.
            \item \textbf{Prod.} topo.: $\mathcal{S} \coloneqq \left\{\pi_{X_{\alpha}}^{-1}\left(U_{\alpha}\right) \colon \alpha \in \Lambda, U_{\alpha} \in \mathcal{T}_{\alpha}\right\}$.
            \item Prod. topo. is the coarsest for cont. $\pi_{\alpha}$'s.
            \item $x \mapsto (x)^{\N}$ is not cont. in the box topo..
            \item $f$ is cont. iff all components are cont..
            \item Subspace topo. of prod. topo. equals prod. topo. of subspace topologies.
            \item Standard topo. on $\R^n$ is the prod. topo. by standard topologies on $\R^{m_i}$'s.
            \item Prod. space basic open set: all but finitely many factors are not $X_i$.
            \item Box topo. is finer: e.g. $\prod_{i \in \N}U_i$.
            \item $d_1 = \sum_{i = 1}^n d_{X_i}\left(x_i, y_i\right)$ and $d_{\infty} = \max d_{X_i}\left(x_i, y_i\right)$ both induce the prod. topo..
            \item $\rho\left(x, y\right) \coloneqq \frac{d\left(x, y\right)}{1 + d\left(x, y\right)}$ is a metric with $\mathrm{diam}\left(X\right) < 1$. 
            \item $D\left(\bm{x}, \bm{y}\right) \coloneqq \sup \left\{\frac{\rho_{i}\left(x_i, y_i\right)}{i} \colon i \in \Z^+\right\}$ is a metric inducing the infinite prod. topo..
            \item \textbf{Quotient map}: surj and $U$ open iff $f^{-1}\left(U\right)$ open.
            \item Closed not open: $x \mapsto x \in [0, 1] ? x : \left(x \in [2, 3] ? x - 1\right)$.
            \item Open not closed: $x \mapsto x \in (0, 1) ? x : \left(x \in (2, 3) ? x - 1\right)$.
            \item Neither: $X \coloneqq \bigl(\left[0, \infty\right) \times \R\bigr) \cup \bigl(\R \times \left\{0\right\}\bigr)$, $(x, y) \mapsto x$. \textbf{But this is a quotient map!}
            \item Both: $\mathrm{id}_X$.
            \item \textbf{Open map}: cont. map from open set to open set.
            \item Surjective cont. $+$ open or closed $=$ quotient map.
            \item Quotient, open and closed maps are preserved under $\circ$.
            \item \textbf{Saturated}: $A = f^{-1}\bigl(f\left(A\right)\bigr)$ or $A = f^{-1}\left(S\right)$.
            \item Surjective cont. $f$ is a quotient map iff $f\left(A\right)$ is open (closed) whenever $A$ is saturated open (closed).
            \item Restriction to saturated sets preserves quotient map.
            \item Homeomorphism: bijective open map.
            \item Embedding: injective homeomorphism onto $f\left(X\right)$.
            \item Isometric: $d_Y\bigl(y_a, y_b\bigr) = d_X\left(a, b\right)$. Isometry if surj.
        \end{itemize}
        \textbf{Countability}
        \begin{itemize}
            \setlength\itemsep{0.1em}
            \item 2nd countable $\implies\exists$ dense countable subset. $\impliedby$ is true if $X$ is metrisable.
            \item $\bigl(\R, \mathcal{P}\left(\R\right)\bigr)$ is 1st but not 2nd countable.
            \item Uncountable co-finite have no countable basis.
            \item $\exists$ nested countable basis $B_1 \subseteq B_2 \subseteq \cdots$.
            \item $x_n \to x \implies x \in \overline{X}$. $\impliedby$ is true if $X$ is 1st countable.
            \item If $f$ is cont., then for any sequence $f\left(x_i\right) \to f\left(x\right)$. The converse is true if $X$ is first countable.
            \item Lindelof: every open cover has a countable sub-cover.
            \item 2nd countable $\implies$ Lindelof. $\impliedby$ if $X$ is metrisable.
        \end{itemize}
        \textbf{Separation}
        \begin{itemize}
            \setlength\itemsep{0.1em}
            \item Co-finite topo. is $T_1$ and $T_2$ if $X$ is finite.
            \item $T_1 \iff \left\{x\right\}$ is closed $\forall x \in X$.
            \item \textbf{Embedding thm}: $\forall x$ and $\forall$ open $U$ s.t. $x \in U$, $\exists\alpha \in \Lambda$ s.t. $f_{\alpha}\left(x\right) > 0$ and $f_{\alpha}\left(X \setminus U\right) = \left\{0\right\}$, then $F\left(x\right) = \bigl(f_{\alpha}\left(x\right)\bigr)_{\alpha \in \Lambda}$ is embedding.
            \item Metric spaces are $T_2$ so all finite subsets are closed.
            \item $T_2$ iff diagonal is closed.
            \item $T_3 \iff \forall x$ and $\forall$ open $U$ s.t. $x \in U$, $\exists$ open $V$ s.t. $x \in V$ and $\overline{V} \subseteq U$.
            \item $T_4 \iff \forall$ closed $A$ and open $U$ s.t. $A \subseteq U$, $\exists$ open $V$ s.t. $A \subseteq V$ and $\overline{V} \subseteq U$.
            \item $T_4 \implies T_3 \implies T_2 \implies T_1$.
            \item Every metric space is $T_4$.
            \item Every 2nd countable $T_3$ space is $T_4$.
            \item Cont. $f$ separates $A, B$: $f\left(A\right) = \left\{0\right\}$, $f\left(B\right) = \left\{1\right\}$.
            \item \textbf{Urysohn's lem}: Every $T_4$ space is completely normal.
            \item \textbf{Urysohn's metrisation thm}: every 2nd countable $T_3$ space is metrisable.
        \end{itemize}
        \textbf{Compactness}
        \begin{itemize}
            \setlength\itemsep{0.1em}
            \item $\left[-\sqrt{2}, \sqrt{2}\right] \cap \Q$: closed and bounded, not cpt.
            \item Closed subset of cpt space is cpt.
            \item Cpt subset of $T_2$ space is closed.
            \item Set in co-finite space is cpt but closed iff it's finite.
            \item Closure of particular point is not cpt.
            \item Cpt $\implies (T_4 \iff T_3 \iff T_2)$
            \item Cont. $f$ maps cpt set to cpt set.
            \item \textbf{Tube lemma}: If $Y$ is cpt and $N \subseteq X \times Y$ is open and contains $\left\{x_0\right\} \times Y$, then $\exists W \supseteq \left\{x_0\right\}$ open s.t. $W \times Y \subseteq N$.
            \item $Y$ is cpt $T_2 \implies \Gamma_f$ closed $\iff f$ is cont..
            \item $X$ cpt iff $\forall$ set of closed sets $\mathcal{G}$ with f.i.p, $\bigcap\mathcal{G} \neq \varnothing$.
            \item $x$ is \textbf{isolated} $\iff \left\{x\right\}$ is open.
            \item If $U \neq \varnothing$ is open in a $T_2$ space and $x$ is not isolated, then $\exists$ non-empty open $V \subseteq U$ s.t. $x \notin \overline{V}$.
            \item Non-empty $T_2$ space $\approx \R$ if has no isolated point.
            \item \textbf{Tychonoff's thm}: prod. of cpt is cpt.
            \item \textbf{Lim pt cpt}: every infinite $Y \subseteq X$ has a lim pt in $X$. Cpt $\implies$ lim pt cpt.
            \item Let $Y$ be trivial and $\Z^+$ be discrete, then $Y \times \Z^+$ is lim pt cpt but not cpt.
            \item \textbf{Seq cpt}: every sequence has a convergent subsequence. Seq cpt $\implies$ lim pt cpt.
            \item Metric space: cpt iff lim pt cpt iff seq cpt
            \item $\R$ with basis $\left(a, \infty\right)$'s is seq cpt but not lim pt cpt.
            \item \textbf{Lebesgue number}: $\forall S \subseteq X$ with $\mathrm{diam}\left(S\right) < \delta$, $\exists U \in \mathcal{U}$ s.t. $S \subseteq U$. 
            \item Every open cover of seq cpt metric space has a Lebesgue number.
            \item \textbf{Totally bounded}: $\forall \epsilon > 0$, $\exists$ finite cover by $B_{\epsilon}\left(x_i\right)$.
            \item Every seq cpt metric space is totally bounded.
            \item In a metric space, cpt iff lim pt cpt iff seq cpt.
            \item $f \colon X \to Y$ is cont. between metric spaces. If $X$ is cpt, then $f$ is uniformly cont..
            \item Totally bounded $\implies \mathrm{diam}\left(X\right) < \infty$.
            \item $X \subseteq \left(\R^n, d_p\right)$ is totally bounded iff $X$ is bounded.
            \item $\left(1 + \frac{1}{n}\right)^n$ converges to $\e$.
            \item \textbf{Complete}: every Cauchy seq is convergent.
            \item $\left(\R^{\omega}, D\right)$ is complete.
            \item Subspace of complete metric space is complete iff closed.
            \item Cpt iff complete and totally bounded.
            \item \textbf{Heine-Borel Thm}: cpt in $\R^n$ iff closed and bounded.
            \item \textbf{Locally cpt}: $\exists$ cpt $K$ and open $U$ s.t. $x \in U \subseteq K$.
            \item $X$ is locally cpt $T_2$ iff $\exists$ cpt $T_2$ $Y$ and $h_Y \colon X \to Y$ homeomorphic onto $h_Y\left(X\right)$ and $Y = h_Y\left(X\right) \cup \left\{p\right\}$.
            \item Open sets in $Y$ are open sets in $X$ and complements of cpt sets in $X$. $h_Y$ is just the identity.
            \item $\exists$ homeomorphism $f$ s.t. $f\restriction_{h_Y\left(X\right)} = h_{Y'} \circ h_{Y}^{-1}\restriction_{h_Y\left(X\right)}$
            \item $\overline{\mathbb{D}}$ and $\mathbb{S}^2$ are the one-point cptification for $\mathbb{D}$ and $\R^n$ because $\bm{v} \mapsto \frac{\bm{v}}{1 + \norm{\bm{v}}}$ is bijection.
            \item A $T_2$ space is locally cpt iff $\forall x$ and open $U$ s.t. $x \in U$, $\exists$ open cpt $V$ s.t. $x \in V \subseteq U$.
            \item $X$ locally cpt, then $A \subseteq X$ locally cpt if $A$ closed or $A$ open and $X$ $T_2$.
            \item $X$ locally cpt $T_2$ iff $\exists$ cpt $T_2$ $Y$ s.t. $X$ homeomorphic to open $A \subseteq Y$.
            \item \textbf{Uniform metric}: $\overline{\rho}\left(f, g\right) = \sup\left\{\rho\bigl(\pi_{\alpha}\left(f\right), \pi_{\alpha}\left(g\right)\bigr)\right\}$.
            \item Prod. $\subseteq$ Uniform $\subseteq$ Box
            \item $Y$ is complete $\implies Y^X$ is complete.
            \item $\mathcal{C}\left(X, Y\right)$ and $\mathcal{B}\left(X, Y\right)$ are closed, and complete if $Y$ is.
            \item \textbf{Supremum metric}: $d_{\sup}\left(f, g\right) = \sup d\bigl(f\left(x\right), g\left(x\right)\bigr)$.
            \item \textbf{Metric completion}: isometric $\phi$ s.t. $\overline{\phi\left(X\right)} = Y$.
            \item Every metric space has a unique metric completion up to isometry s.t. $f\restriction_{\phi\left(X\right)} = \phi' \circ \phi^{-1}$.
            \item \textbf{Topo. of cpt convergence}: gen'ed by $B\left(K, f, \epsilon\right) \coloneqq \left\{g \in Y^X \colon d_{\sup}\left(f\restriction_K, g\restriction_K\right) < \epsilon\right\}$'s.
            \item \textbf{Cpt-open topo.}: gen'ed by $S\left(K, U\right) \coloneqq \left\{g \in \mathcal{C}\left(X, Y\right) \colon g\left(K\right) \subseteq U\right\}$'s.
            \item Both are the same on $\mathcal{C}\left(X, Y\right)$.
            \item \textbf{Cptly gen'ed}: $A$ open (closed) in $X$ iff $A \cap K$ open (closed) in $K$ for all cpt sets in $X$.
            \item Locally cpt or 1st countable $\implies$ cptly gen'ed.
            \item Cptly gen'ed $\implies$ ($f$ cont. iff $f\restriction_K$ cont. $\forall$ cpt $K$).
            \item $X$ cptly gen'ed $\implies \mathcal{C}\left(X, Y\right)$ closed under topo. of cpt convergence.
            \item \textbf{Equicont.}: $\exists$ open $U \forall \epsilon > 0$ s.t. $\forall f \in \mathcal{Y} \subseteq \mathcal{C}\left(X, Y\right)$, $d\left(y_{x_0}, y_x\right) < \epsilon \forall x \in U$.
            \item Totally bounded in $\overline{\rho} \implies$ equicont.
            \item Converge in prod. topo. $=$ ptwise convergence.
            \item Equicont. family has the same topo. for cpt and ptwise convergence.
            \item \textbf{Arzela-Ascoli Thm}: $\mathcal{Y}_a \coloneqq \left\{f\left(a\right) \colon f \in \mathcal{Y}\right\}$. $\overline{\mathcal{Y}} \subseteq \mathcal{C}\left(X, Y\right)$ is cpt if it's equicont. and $\overline{\mathcal{Y}_a}$ is cpt $\forall a \in X$. $\impliedby$ is true if $X$ is locally cpt $T_2$. 
        \end{itemize}
        \textbf{Connectedness}
        \begin{itemize}
            \setlength\itemsep{0.1em}
            \item Connected iff only $X$ and $\varnothing$ are clopen (contrapositive).
            \item $X = \left(U, V\right)$, $Y \subseteq X$ connected $\implies Y \subseteq U$ or $Y \subseteq V$ (contrapositive).
            \item Connected sets with $\cap \neq \varnothing$ has connected $\cup$ (suppose not and condition on $A_{\alpha} \setminus U$).
            \item $A$ connected and $A \subseteq B \subseteq \overline{A} \implies B$ connected (suppose not).
            \item Cont. $f$ maps connected to connected (contrapositive).
            \item Prod. of connected is connected (contrapositive).
            \item Path connected $\implies$ connected (contrapositive, suppose exists cont. $f$).
            \item Path component $\subseteq $ cc.
            \item \textbf{Locally connected}: $\forall$ open $U$ s.t. $x \in U$, $\exists$ open connected $V$ s.t. $x \in V \subseteq U$.
            \item $S$: graph of $\sin\frac{2\pi}{x}$ at $\left(0, 1\right]$. $\overline{S} = S \cup \left\{0\right\} \times \left[-1, 1\right]$ is connected, but not path/locally connected (no $(0, 0)$-$(x, y)$ path: $\max f^{-1}(\left\{0\right\} \times [-1, 1]) \mapsto y_0 \coloneqq \lim y\left(t\right)$; not locally connected: $0.5$-ball at $\mathbf{0}$ contains a $\delta$-square, $\exists x \in (0, \delta)$ s.t. $\sin = 1$, split there).
            \item Locally connected iff every cc of every open set is open.
            \item CCs and path components are the same in locally path connected space (suppose not but a path cc $\subseteq$ a cc).
            \item Quotient topo. on $\tilde{X}$: cc's of locally path connected space, is discrete (prove $\mathcal{P}\left(\tilde{X}\right)$ is a quotient topo.).
            \item Locally path connected $\implies$ locally connected (counter e.g. empty graph).
        \end{itemize}
