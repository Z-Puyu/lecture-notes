        \begin{itemize}
            \item \textbf{strictly dominated strategy}: $u_i(s_i, \bm{s}_{-i}) < u_i(s_i', \bm{s}_{-i})$ for all $\bm{s}_{-i} \in S_{-i}$.
            \item \textbf{best-response correspondence}: $R_i\left(\bm{s}_{-i}\right) \coloneqq \argmax_{s_i \in S_i} u_i\left(s_i, \bm{s}_{-i}\right)$.
            \item \textbf{Nash equilibrium}: $s_i^* \in R_i\left(\bm{s}^*_{-i}\right) \iff u_i\left(s_i^*, \bm{s}_{-i}^*\right) = \max_{s_i \in S_i}u_i\left(s_i, \bm{s}_{-i}^*\right)$ for all $i$.
            \item Define $G\left(R_i\right) \coloneqq \left\{\left(s_i, \bm{s}_{-i}\right) \colon s_i \in R_i\left(\bm{s}_{-i}\right), \bm{s}_{-i} \in S_{-i}\right\}$ as the \textbf{graph} of $R_i$, then $\bigcap_{i = 1}^nG\left(R_i\right)$ are exactly all the Nash equilibria.
            \item $2$-player games: $\left(s_1, s_2\right)$ is a Nash equilibrium iff $s_1 \in R_1\left(s_2\right)$ and $s_2 \in R_2\left(s_1\right)$.
            \item If $\left(s_1^*, s_2^*, \cdots, s_n^*\right)$ are a Nash equilibrium, then no $s_i^*$ can be eliminated by IESDS. \textbf{Proof: suppose $s_j$ is the first to be eliminated.}
            \item If IESDS results in a single set of strategies, they are the unique Nash equilibrium of the game. \textbf{Proof: uniqueness is trivial. For existence, suppose $s_i^* \notin R_i\left(\bm{s}_{-i}^*\right) \neq \varnothing$ for some $i$.}
            \item \textbf{Mixed strategy}: $p_i \coloneqq \left(p_{ij}\right)_{j = 1}^k$ s.t. $P\left(\textrm{Player } i \textrm{ plays } s_{ij}\right) = p_{ij}$.
            \item \textbf{Expected payoff for pure strategy}: 
            \begin{equation*}
                v_i\left(x, p_j\right) \coloneqq \sum_{n = 1}^{N}p_{jn}u_i\left(x, s_{jn}\right).
            \end{equation*}
            \item \textbf{Expected payoff for mixed strategy}: 
            \begin{equation*}
                v_i\left(p_i, p_j\right) \coloneqq \sum_{m = 1}^{M}p_{im}\sum_{n = 1}^{N}p_{jn}u_i\left(s_{im}, s_{jn}\right).
            \end{equation*}
            \item If $s_{ik}$ is eliminated by IESDS, then $p_{ik}^* = 0$.
            \item \textbf{Nash's Theorem}: at least one Nash equilibrium exists for any finite game if mixed strategies are allowed.
            \item \textbf{Market clearing price}: the price at which the demand is equal to the supply.
            \item \textbf{Demand-side reservation price}: the lowest price at which the demand is $0$.
            \item \textbf{Cournot Model of Duopoly}:
            \begin{itemize}
                \item Demand: $Q\left(P\right) = ka - kP$ where $a$ is the reservation price.
                \item Price: $P\left(Q\right) = \begin{cases}
                    0 & \quad\textrm{if } Q \geq ka \\
                    a - \frac{1}{k}Q & \quad\textrm{otherwise}
                \end{cases}$.
                \item Profit: $\pi_i\left(q_i, q_j\right) = \begin{cases}
                    \left(a - \frac{1}{k}q_i - \frac{1}{k}q_j - c\right)q_i & \quad\textrm{if } Q < ka \\
                    -cq_i & \quad\textrm{if } Q \geq ka
                \end{cases}$.
                \item Best response: $R_i\left(q_j\right) = \begin{cases}
                    \frac{1}{2}\left(ka - q_j - kc\right) & \quad\textrm{if } q_2 \leq k\left(a - c\right) \\
                    0 & \quad\textrm{otherwise}
                \end{cases}$.
            \end{itemize}
            \item \textbf{Bertrand Model of Duopoly}:
            \begin{itemize}
                \item Demand: $q_i\left(p_i, p_j\right) = ka - kp_i + bp_j$ where $a$ is the reservation price and $b$ is the substitutability coefficient.
                \item Profit: $\pi_i\left(p_i, p_j\right) = \left(ka - kp_i + bp_2\right)\left(p_i - c\right)$.
                \item Best response: $R_i\left(p_j\right) = \frac{1}{2k}\left(ka + bp_j + kc\right)$.
                \item Nash equilibrium: $p_1^* = p_2^* = \frac{ka + kc}{2k - b}$, i.e., the model makes sense iff $b > 2k$.
                \item Homogeneous product Nash equilibrium: $p_1^* = p_2^* = c$.
            \end{itemize}
            \item \textbf{Problem of the commons}:
            \begin{itemize}
                \item Unit utility given total consumption: $v\left(x\right)$ is \textbf{strictly concave} for $x < X_{\max}$ and $v\left(x\right) = 0$ for $x \geq X_{\max}$.
                \item Payoff: $u_i\left(x_i, \bm{x}_{-i}\right) \coloneqq x_iv\left(X\right) - cx_i$.
                \item Nash equilibrium solves $v\left(X^*\right) + x_i^*\left.\frac{\mathrm{d} v}{\mathrm{d} x_i}\right\rvert_{x_i = x_i^*} - c = 0$.
                \item If $v\left(x\right) = a - x^2$ for $x < X_{\max}$, then $x_i^* = \sqrt{\frac{a - c}{n^2 + 2n}}$.
                \item Social welfare: $w\left(X\right) \coloneqq Xv\left(X\right) - Xc$.
                \item If $x^*$ optimises $w$, then $x^* < X^*$. \textbf{Proof: if $x^* \geq X^*$, then $v\left(X^*\right) \geq v\left(x^*\right)$ and $0 > v'\left(X^*\right) \geq v'\left(x^*\right)$. Use this to show $v\left(X^*\right) + \frac{X^*}{n}v'\left(X^*\right) > v\left(x^*\right) + x^*v'\left(x^*\right)$. Use stationary points to show $w'\left(x^*\right) = 0$ and $u_i'\left(x_i^*\right) = 0$.}
            \end{itemize}
            \item \textbf{Final-offer arbitration}:
            \begin{itemize}
                \item Union offers $w_u$ and firm offers $w_f$.
                \item Arbitrator expectation $x \sim f$ with $P\left(x < x_m\right) = F\left(x_m\right) = \frac{1}{2}$.
                \item Expected wage: $\phi\left(w_f, w_u\right) = \mathbb{E}\left[w\right] = w_fF\left(\frac{w_f + w_u}{2}\right) + w_u\biggl(1 - F\left(\frac{w_f + w_u}{2}\right)\biggr)$.
                \item Best responses: 
                \begin{equation*}
                    R_f\left(w_u\right) = \argmin_{0 \leq w_f < w_u^*}\phi\left(w_f, w_u^*\right), \qquad R_u\left(w_f\right) = \argmax_{w_u > w_f^*}\phi\left(w_f^*, w_u\right).
                \end{equation*} 
                \item Nash equilibrium: $w_f^* = x_m - \frac{1}{2f\left(x_m\right)}$, $w_u^* = x_m + \frac{1}{2f\left(x_m\right)}$.
            \end{itemize}
            \item \textbf{Backward induction}: 
            \begin{itemize}
                \item Define $b_j\left(b_1\right) \coloneqq R_j\left(b_1, \cdots, b_{j - 1}\right)$ as player $j$'s best action given player $1$'s action.
                \item $a_i^* \coloneqq \argmax_{a_1 \in A_1}u_1\bigl(a_1, b_2\left(a_1\right), \cdots, b_n\left(a_1\right)\bigr)$ and $a_i^* \coloneqq b_i\left(a_1^*\right)$.
            \end{itemize}
            \item \textbf{Extensive-form game} specifies:
            \begin{enumerate}
                \item the players in the game;
                \item when each player takes a move;
                \item each player's strategy space;
                \item each player's knowledge of the game;
                \item the payoffs for each possible combination of strategies.
            \end{enumerate}
            \item \textbf{Information set}: a set of \textbf{decision nodes} joined with \textbf{dotted line in a game tree} for the same player at a turn s.t. the player cannot deduce the predecessor node. \textbf{Perfect information} $\iff$ all information sets are singletons.      
            \item \textbf{Subgame}: a game based on a subtree of a game tree s.t. its decision nodes can be partitioned by information sets.
            \item \textbf{Strategy in dynamic games}: a map $s_i \colon \prod_{j = 1}^{i - 1}A_j \to A_i$.
            \item Action of player $i$ following strategies $\bm{s} \coloneqq \left(s_j\right)_{j = 1}^n$ is $a_i\left(\bm{s}\right) = s_i\bigl(a_1\left(\bm{s}\right), \cdots, a_{i - 1}\left(\bm{s}\right)\bigr)$.
            \item \textbf{Payoff in dynamic games}: $\tilde{\bm{u}}\left(\bm{s}\right) \coloneqq \bm{u}\bigl(a_1\left(\bm{s}\right), a_2\left(\bm{s}\right), \cdots, a_n\left(\bm{s}\right)\bigr)$.
            \item $\bigl(a_1^*, R_2\left(a_1^*\right)\bigr)$ is a backward induction outcome $\implies \left(a_1^*, R_2\right)$ is a Nash equilibrium. \textbf{Converse is not true in general}.
            \item \textbf{Subgame perfect Nash equilibrium} is a Nash equilibrium in every subgame.
            \item \textbf{Stackelberg Model of Duopoly}:
            \begin{itemize}
                \item Dominate firm $1$ and follower firm $2$.
                \item Price: $P\left(Q\right) = \begin{cases}
                    a - \frac{1}{k}Q & \quad\textrm{if } Q < ka \\
                    0 & \quad\textrm{if } Q \geq ka
                \end{cases}$.
                \item $R_2\left(q_1\right) = \begin{cases}
                    \frac{1}{2}\left(ka - q_1 - kc\right) & \quad\textrm{if } q_1 < k\left(a - c\right) \\
                    0 & \quad\textrm{otherwise}
                \end{cases}$.
            \end{itemize}
            \item \begin{tabular}{|c|c|c|}
                \hline 
                & Cournot & Stackelberg \\
                \hline
                $\left(q_1^*, q_2^*\right)$ & $\left(\frac{ka - kc}{3}, \frac{ka - kc}{3}\right)$ & $\left(\frac{ka - kc}{2}, \frac{ka - kc}{4}\right)$ \\
                \hline
                $P$ & $c + \frac{a - c}{3}$ & $c + \frac{a - c}{4}$ \\
                \hline
                $\left(\pi_1^*, \pi_2^*\right)$ & $\left(\frac{k^2\left(a - c\right)^2}{9}, \frac{k^2\left(a - c\right)^2}{9}\right)$ & $\left(\frac{k^2\left(a - c\right)^2}{8}, \frac{k^2\left(a - c\right)^2}{16}\right)$ \\
                \hline
                $\pi_1^* + \pi_2^*$ & $\frac{2k^2\left(a - c\right)^2}{9}$ & $\frac{3k^2\left(a - c\right)^2}{16}$ \\
                \hline
            \end{tabular}
            \item \textbf{Three-period bargaining}: if the players will receive $\left(u_1, u_2\right)$ in period $i + 1$, then player $i$ should offer $\delta u_j$ and player $j$ should accept.
            \item In a three-period bargaining game, player $1$ will offer $\begin{cases}
                    s_1\left(1\right) & = 1 - \delta\left(1 - \delta\bar{s}_1\right) \\
                    s_2\left(1\right) & = \delta\left(1 - \delta\bar{s}_1\right)
                \end{cases}$
            and player $2$ will accept the offer.
            \item \textbf{Infinite-horizon bargaining}: if $\left(\bar{s}_1, \bar{s}_2\right)$ are final offers received in backward induction, then by collapsing stage $3$ onwards, the game is equivalent to a $3$-period bargaining with $ \begin{cases}
                \bar{s}_1 & = 1 - \delta\left(1 - \delta\bar{s}_1\right) \\
                \bar{s}_2 & = \delta\left(1 - \delta\bar{s}_1\right)
            \end{cases}$ which yields $\bar{s}_1 = \frac{1}{1 + \delta}$.
            \item \textbf{Multi-stage game}: dynamic games which can be partitioned into static stage games. The game is \textbf{repeated} if every stage is the same.
            \item If the stage game of a repeated game has a unique Nash equilibrium, then the repeated game has a unique subgame perfect Nash equilibrium.
            \item \textbf{Present value}: $\sum_{i = 1}^{\infty}\delta^{i - 1}\pi_i$ where $\pi_i$ is the stage $i$ payoff and $\delta \in \left(0, 1\right)$ is the \textbf{discount factor}.
            \item If the stage game with left (not cooperate) and right (cooperate) strategies is 
            \begin{equation*}
                \begin{bmatrix}
                    & $L$ & $R$ \\
                    $L$ & \left(1, 1\right) & \left(5, 0\right) \\
                    $R$ & \left(0, 5\right) & \left(4, 4\right)
                \end{bmatrix},
            \end{equation*}
            then $\left(NC, NC\right)$ is always a Nash equilibrium and $\left(T, T\right)$ is a Nash equilibrium iff $\delta \geq \frac{1}{4}$. \textbf{Proof: show that $T$ is the best response for $T$, i.e., if player $1$ plays $T$, then player $2$ will end up playing $T$.}
        \end{itemize}
